// ================================
// Firestore Rules
// ================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function verified() {
      return signedIn() && (
        request.auth.token.email_verified == true ||
        (
          exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.emailVerified == true
        )
      );
    }

    function validUsername(u) {
      return u is string && u.matches('^[A-Za-z0-9_]{3,20}$');
    }

    function validLowerUsername(u) {
      return u is string && u.matches('^[a-z0-9_]{3,20}$');
    }

    function validPairKey(uidA, uidB, pairKey) {
      return pairKey == uidA + '_' + uidB || pairKey == uidB + '_' + uidA;
    }

    function authUidInPairKey(pairKey) {
      return signedIn() && (
        pairKey.matches('^' + request.auth.uid + '_[A-Za-z0-9]+$') ||
        pairKey.matches('^[A-Za-z0-9]+_' + request.auth.uid + '$')
      );
    }

    function isChatParticipant(pairKey) {
      return signedIn()
        && exists(/databases/$(database)/documents/chats/$(pairKey))
        && request.auth.uid in get(/databases/$(database)/documents/chats/$(pairKey)).data.participants;
    }

    match /usernames/{usernameLower} {
      allow get: if true;
      allow list: if false;

      allow create: if signedIn()
        && validLowerUsername(usernameLower)
        && request.resource.data.keys().hasOnly(['uid', 'username', 'usernameLower', 'createdAt'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.usernameLower == usernameLower
        && validUsername(request.resource.data.username)
        && !exists(/databases/$(database)/documents/usernames/$(usernameLower));

      allow update, delete: if false;
    }

    match /users/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || verified());
      allow list: if false;

      allow create: if signedIn()
        && request.auth.uid == uid
        && request.resource.data.uid == uid
        && validUsername(request.resource.data.username)
        && validLowerUsername(request.resource.data.usernameLower)
        && existsAfter(/databases/$(database)/documents/usernames/$(request.resource.data.usernameLower))
        && getAfter(/databases/$(database)/documents/usernames/$(request.resource.data.usernameLower)).data.uid == uid;

      allow update: if signedIn()
        && request.auth.uid == uid
        && request.resource.data.uid == uid
        && request.resource.data.username == resource.data.username
        && request.resource.data.usernameLower == resource.data.usernameLower;

      allow delete: if false;
    }

    match /requests/{pairKey} {
      allow get: if authUidInPairKey(pairKey);
      allow list: if signedIn() &&
        (resource.data.fromUid == request.auth.uid || resource.data.toUid == request.auth.uid);

      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['fromUid', 'toUid', 'fromUsername', 'toUsername', 'status', 'createdAt', 'updatedAt'])
        && request.resource.data.fromUid == request.auth.uid
        && request.resource.data.toUid is string
        && request.resource.data.toUid != request.auth.uid
        && validPairKey(request.resource.data.fromUid, request.resource.data.toUid, pairKey)
        && validUsername(request.resource.data.fromUsername)
        && validUsername(request.resource.data.toUsername)
        && request.resource.data.status == 'pending'
        && !exists(/databases/$(database)/documents/chats/$(pairKey));

      allow update: if signedIn()
        && resource.data.toUid == request.auth.uid
        && resource.data.status == 'pending'
        && request.resource.data.fromUid == resource.data.fromUid
        && request.resource.data.toUid == resource.data.toUid
        && request.resource.data.fromUsername == resource.data.fromUsername
        && request.resource.data.toUsername == resource.data.toUsername
        && request.resource.data.status in ['accepted', 'rejected'];

      allow delete: if false;
    }

    match /chats/{pairKey} {
      allow get: if authUidInPairKey(pairKey) ||
        (signedIn() && request.auth.uid in resource.data.participants);

      allow list: if signedIn() && request.auth.uid in resource.data.participants;

      allow create: if signedIn()
        && request.resource.data.participants is list
        && request.resource.data.participants.size() == 2
        && request.auth.uid in request.resource.data.participants
        && validPairKey(request.resource.data.participants[0], request.resource.data.participants[1], pairKey)
        && existsAfter(/databases/$(database)/documents/requests/$(pairKey))
        && getAfter(/databases/$(database)/documents/requests/$(pairKey)).data.status == 'accepted';

      allow update: if signedIn()
        && request.auth.uid in resource.data.participants
        && request.resource.data.participants == resource.data.participants;

      allow delete: if false;

      match /messages/{messageId} {
        allow get, list: if isChatParticipant(pairKey);

        allow create: if isChatParticipant(pairKey)
          && request.resource.data.keys().hasOnly(['senderUid', 'senderUsername', 'text', 'createdAt', 'seenBy'])
          && request.resource.data.senderUid == request.auth.uid
          && request.resource.data.senderUsername is string
          && validUsername(request.resource.data.senderUsername)
          && request.resource.data.text is string
          && request.resource.data.text.size() > 0
          && request.resource.data.text.size() <= 1000
          && request.resource.data.seenBy is map
          && request.resource.data.seenBy.keys().hasOnly([request.auth.uid])
          && request.resource.data.seenBy[request.auth.uid] == true;

        allow update: if isChatParticipant(pairKey)
          && request.resource.data.keys().hasOnly(['senderUid', 'senderUsername', 'text', 'createdAt', 'seenBy'])
          && request.resource.data.senderUid == resource.data.senderUid
          && request.resource.data.senderUsername == resource.data.senderUsername
          && request.resource.data.text == resource.data.text
          && request.resource.data.createdAt == resource.data.createdAt
          && request.resource.data.seenBy is map
          && request.resource.data.seenBy[request.auth.uid] == true
          && request.resource.data.seenBy.keys().hasOnly([resource.data.senderUid, request.auth.uid])
          && request.resource.data.seenBy.diff(resource.data.seenBy).affectedKeys().hasOnly([request.auth.uid]);

        allow delete: if isChatParticipant(pairKey)
          && request.auth.uid == resource.data.senderUid;
      }
    }

    match /statuses/{ownerUid} {
      allow get, list: if signedIn()
        && (
          ownerUid == request.auth.uid ||
          request.auth.uid in resource.data.audienceUids
        );

      allow create, update: if signedIn()
        && ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly([
          'ownerUid',
          'ownerUsername',
          'ownerProfilePhotoUrl',
          'imageUrl',
          'audienceUids',
          'createdAt',
          'updatedAt',
          'expiresAtMs'
        ])
        && request.resource.data.ownerUid == request.auth.uid
        && validUsername(request.resource.data.ownerUsername)
        && request.resource.data.ownerProfilePhotoUrl is string
        && request.resource.data.imageUrl is string
        && request.resource.data.imageUrl.size() > 0
        && request.resource.data.audienceUids is list
        && request.auth.uid in request.resource.data.audienceUids
        && (request.resource.data.expiresAtMs is int || request.resource.data.expiresAtMs is float)
        && request.resource.data.expiresAtMs > request.time.toMillis();

      allow delete: if signedIn() && request.auth.uid == ownerUid;
    }
  }
}
